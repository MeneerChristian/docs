  Where the source code has an XML documentation comment for a type or member, its summary (and the exceptions it documents) is shown as a comment above the declaration.

  for example, `AiAgentTruncateChat` is public, but the only constructor and property of the client that accept it are `internal`.

// Records the statically-known shape of a single index class so the RavenDB analyzers can reason about it from
//    any assembly. Emitted per index type at assembly level by the analyzer package's source generator; it is
//    not meant to be written by hand.
public sealed class RavenIndexMetadataAttribute : Attribute
{
    RavenIndexMetadataAttribute(Type indexType);
    // Number of AddMap / AddMapForAll call sites across the constructors in the chain.
    int AddMapCount { get; set; }
    // Whether any of those AddMap call sites sits inside a loop, in which case the call count understates how many
    //    maps are registered at runtime.
    bool AddMapInLoop { get; set; }
    // Whether the generator could read the index completely. False when the shape cannot be determined statically:

    bool Analyzable { get; set; }
    // Whether any constructor in the chain assigns the Map property.
    bool AssignsMap { get; set; }
    // Whether 'MapFields', 'StoredFields' and 'StoreAllFields' could be read. False when the map projection or a

    //    index defines itself rather than which fields it produces.
    bool FieldsKnown { get; set; }
    // The index class this metadata describes.
    Type IndexType { get; }
    // Whether any constructor in the chain assigns the Reduce property. A map-reduce index serves every projected

    bool IsMapReduce { get; set; }

    string[] MapFields { get; set; }

    bool StoreAllFields { get; set; }

    string[] StoredFields { get; set; }
    // Whether the chain writes to AdditionalSources or AdditionalAssemblies, shipping C# for the server to compile.

    //    stand down for such an index.
    bool UsesAdditionalCode { get; set; }
